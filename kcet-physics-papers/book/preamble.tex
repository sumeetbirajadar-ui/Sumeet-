\newcommand{\pageframe}{%
  \begin{tikzpicture}[remember picture,overlay]
    \coordinate (SW) at ($(current page.south west)+(0.85cm,0.85cm)$);
    \coordinate (NE) at ($(current page.north east)+(-0.85cm,-0.85cm)$);
    \coordinate (SWi) at ($(SW)+(0.16cm,0.16cm)$);
    \coordinate (NEi) at ($(NE)-(0.16cm,0.16cm)$);
    \draw[gilt,line width=1.8pt] (SW) rectangle (NE);
    \draw[navy,line width=0.5pt] (SWi) rectangle (NEi);
    % corner ornaments
    \node[anchor=north west,inner sep=0pt] at ($(NEi -| SWi)+(0.06,-0.06)$)
         {\pgfornament[width=1.25cm,color=gilt]{61}};
    \node[anchor=north east,inner sep=0pt] at ($(NEi)+(-0.06,-0.06)$)
         {\pgfornament[width=1.25cm,color=gilt,symmetry=v]{61}};
    \node[anchor=south west,inner sep=0pt] at ($(SWi)+(0.06,0.06)$)
         {\pgfornament[width=1.25cm,color=gilt,symmetry=h]{61}};
    \node[anchor=south east,inner sep=0pt] at ($(SWi -| NEi)+(-0.06,0.06)$)
         {\pgfornament[width=1.25cm,color=gilt,symmetry=c]{61}};
    % gold medallion bands centred on top and bottom rules
    \foreach \yy in {0,1}{
      \ifnum\yy=0 \coordinate (M) at ($(NE -| current page.center)$);
